Toast 提示

操作回饋提示，從右下角滑入。使用 JavaScript 動態建立。
檔案來源：design-system.css → .toast-v2

JavaScript

function showToast(message, type = 'info') {
    const existing = document.querySelector('.toast-v2');
    if (existing) existing.remove();

    const toast = document.createElement('div');
    toast.className = `toast-v2 toast-v2--${type}`;
    const icons = {
        success: 'check-circle',
        error: 'exclamation-circle',
        warning: 'exclamation-triangle'
    };
    toast.innerHTML = `<i class="fas fa-${icons[type]}"></i> ${message}`;
    document.body.appendChild(toast);

    setTimeout(() => {
        toast.style.animation = 'slideInRight 0.3s reverse forwards';
        setTimeout(() => toast.remove(), 300);
    }, 3000);
}

粒子吸引效果 Particle Attraction

未登入頁面專用的背景粒子效果。粒子會緩慢浮動，游標接近時被吸引。
檔案來源：assets/js/particle-attraction.js

移動游標試試看

JavaScript

// 引入 particle-attraction.js 後自動初始化
// 僅在 desktop + body.landing-page 時啟用

<script src="assets/js/particle-attraction.js"></script>

// 或手動初始化：
new ParticleAttraction({
    particleCount: 60,    // 粒子數量
    attractionRadius: 200, // 吸引範圍 (px)
    attractionForce: 0.12, // 吸引力
    friction: 0.93,        // 摩擦力
    returnForce: 0.03      // 回歸力
});
